A **Browser Profile** is a saved snapshot of browser state—cookies, localStorage, and session files—that you can reuse across multiple runs. Profiles let you skip login steps and restore authenticated state instantly.

Profiles are ideal when you:
- Run the same workflow repeatedly with the same account (daily data extraction, scheduled reports)
- Want multiple workflows to share the same authenticated state
- Need to avoid repeated authentication to save time and steps

---

## Create a Browser Profile

There are 2 ways to create browser profiles:

1. Run a workflow session with `persist_browser_session=True`
2. Create a profile from an closed browser session

### Method A: From a workflow run

Run a workflow with `persist_browser_session=True`, wait for it to complete, then create a profile from the run. Session archiving happens asynchronously, so add brief retry logic when creating the profile.

<CodeGroup>
```python Python
import asyncio
from skyvern import Skyvern

async def main():
    client = Skyvern(api_key="YOUR_API_KEY")

    # Step 1: Run workflow with persist_browser_session=True
    workflow_run = await client.run_workflow(
        workflow_id="wf_login_to_dashboard",
        persist_browser_session=True,
        wait_for_completion=True,
    )

    print(f"Workflow completed: {workflow_run.status}")

    # Step 2: Create profile from the completed run
    # Retry briefly while session archives asynchronously
    for attempt in range(10):
        try:
            profile = await client.browser_profiles.create_browser_profile(
                name="analytics-dashboard-login",
                workflow_run_id=workflow_run.run_id,
                description="Authenticated state for analytics dashboard",
            )
            print(f"Profile created: {profile.browser_profile_id}")
            break
        except Exception as e:
            if "persisted" in str(e).lower() and attempt < 9:
                await asyncio.sleep(1)
                continue
            raise

asyncio.run(main())
```

```typescript TypeScript
import { SkyvernClient } from "@skyvern/client";

async function waitForRun(client: SkyvernClient, runId: string) {
  while (true) {
    const run = await client.getRun(runId);
    if (["completed", "failed", "terminated", "timed_out", "canceled"].includes(run.status)) {
      return run;
    }
    await new Promise((r) => setTimeout(r, 5000));
  }
}

async function main() {
  const client = new SkyvernClient({ apiKey: process.env.SKYVERN_API_KEY });

  // Step 1: Run workflow with persist_browser_session=true
  const workflowRun = await client.runWorkflow({
    workflowId: "wf_login_to_dashboard",
    persistBrowserSession: true,
  });

  const completedRun = await waitForRun(client, workflowRun.runId);
  console.log(`Workflow completed: ${completedRun.status}`);

  // Step 2: Create profile from the completed run
  // Retry briefly while session archives asynchronously
  let profile;
  for (let attempt = 0; attempt < 10; attempt++) {
    try {
      profile = await client.browserProfiles.createBrowserProfile({
        name: "analytics-dashboard-login",
        workflowRunId: completedRun.runId,
        description: "Authenticated state for analytics dashboard",
      });
      break;
    } catch (e) {
      if (String(e).toLowerCase().includes("persisted") && attempt < 9) {
        await new Promise((r) => setTimeout(r, 1000));
        continue;
      }
      throw e;
    }
  }

  console.log(`Profile created: ${profile.browserProfileId}`);
}

main();
```

```bash cURL
# Step 1: Run workflow with persist_browser_session=true
WORKFLOW_RESPONSE=$(curl -s -X POST "https://api.skyvern.com/v1/run/workflows" \
  -H "x-api-key: $SKYVERN_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{
    "workflow_id": "wf_login_to_dashboard",
    "persist_browser_session": true
  }')

RUN_ID=$(echo "$WORKFLOW_RESPONSE" | jq -r '.run_id')

# Wait for completion
while true; do
  STATUS=$(curl -s "https://api.skyvern.com/v1/runs/$RUN_ID" \
    -H "x-api-key: $SKYVERN_API_KEY" | jq -r '.status')
  if [ "$STATUS" != "created" ] && [ "$STATUS" != "running" ]; then
    break
  fi
  sleep 5
done

# Step 2: Create profile (retry while session archives)
for i in {1..10}; do
  PROFILE_RESPONSE=$(curl -s -X POST "https://api.skyvern.com/v1/browser_profiles" \
    -H "x-api-key: $SKYVERN_API_KEY" \
    -H "Content-Type: application/json" \
    -d "{
      \"name\": \"analytics-dashboard-login\",
      \"workflow_run_id\": \"$RUN_ID\",
      \"description\": \"Authenticated state for analytics dashboard\"
    }")
  if echo "$PROFILE_RESPONSE" | jq -e '.browser_profile_id' > /dev/null 2>&1; then
    echo "$PROFILE_RESPONSE"
    break
  fi
  sleep 1
done
```
</CodeGroup>

**Parameters:**

| Parameter | Type | Description |
|-----------|------|-------------|
| `name` | string | Required. Display name for the profile. Must be unique within your organization |
| `workflow_run_id` | string | Required. ID of the completed workflow run (starts with `wr_`) |
| `description` | string | Optional description of the profile's purpose |

### Method B: From a browser session

Run tasks in a browser session, close the session, then create a profile from it.

<CodeGroup>
```python Python
import asyncio
from skyvern import Skyvern

async def main():
    client = Skyvern(api_key="YOUR_API_KEY")

    # Step 1: Create a browser session
    session = await client.create_browser_session(timeout=30)
    session_id = session.browser_session_id

    # Step 2: Run tasks in the session
    await client.run_task(
        prompt="Login with username 'admin@company.com'",
        url="https://dashboard.example.com/login",
        browser_session_id=session_id,
        wait_for_completion=True,
    )

    # Step 3: Close the session
    await client.close_browser_session(browser_session_id=session_id)

    # Step 4: Create profile from the closed session
    profile = await client.browser_profiles.create_browser_profile(
        name="dashboard-admin-login",
        browser_session_id=session_id,
        description="Admin account for dashboard access",
    )

    print(f"Profile created: {profile.browser_profile_id}")

asyncio.run(main())
```

```bash cURL
# Step 1: Create a browser session
SESSION_RESPONSE=$(curl -s -X POST "https://api.skyvern.com/v1/browser_sessions" \
  -H "x-api-key: $SKYVERN_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{"timeout": 30}')

SESSION_ID=$(echo "$SESSION_RESPONSE" | jq -r '.browser_session_id')

# Step 2: Run a task in the session
TASK_RESPONSE=$(curl -s -X POST "https://api.skyvern.com/v1/run/tasks" \
  -H "x-api-key: $SKYVERN_API_KEY" \
  -H "Content-Type: application/json" \
  -d "{
    \"prompt\": \"Login with username 'admin@company.com'\",
    \"url\": \"https://dashboard.example.com/login\",
    \"browser_session_id\": \"$SESSION_ID\"
  }")

RUN_ID=$(echo "$TASK_RESPONSE" | jq -r '.run_id')

# Wait for task to complete
while true; do
  STATUS=$(curl -s "https://api.skyvern.com/v1/runs/$RUN_ID" \
    -H "x-api-key: $SKYVERN_API_KEY" | jq -r '.status')
  if [ "$STATUS" != "created" ] && [ "$STATUS" != "running" ] && [ "$STATUS" != "queued" ]; then
    break
  fi
  sleep 5
done

# Step 3: Close the session
curl -s -X POST "https://api.skyvern.com/v1/browser_sessions/$SESSION_ID/close" \
  -H "x-api-key: $SKYVERN_API_KEY"

# Step 4: Create profile from the closed session
curl -X POST "https://api.skyvern.com/v1/browser_profiles" \
  -H "x-api-key: $SKYVERN_API_KEY" \
  -H "Content-Type: application/json" \
  -d "{
    \"name\": \"dashboard-admin-login\",
    \"browser_session_id\": \"$SESSION_ID\",
    \"description\": \"Admin account for dashboard access\"
  }"
```

## Use a Browser Profile

Pass `browser_profile_id` to any workflow or task to restore the saved state. Skyvern restores cookies, localStorage, and session files before the first step runs.

### In workflows

<CodeGroup>
```python Python
import asyncio
from skyvern import Skyvern

async def main():
    client = Skyvern(api_key="YOUR_API_KEY")

    # Run workflow with saved profile—no login needed
    result = await client.run_workflow(
        workflow_id="wf_daily_metrics",
        browser_profile_id="bp_490705123456789012",
        wait_for_completion=True,
    )

    print(f"Output: {result.output}")

asyncio.run(main())
```


### In tasks

<CodeGroup>
```python Python
import asyncio
from skyvern import Skyvern

async def main():
    client = Skyvern(api_key="YOUR_API_KEY")

    # Run task with saved profile—starts already logged in
    result = await client.run_task(
        prompt="Download the monthly report from the reports section",
        url="https://dashboard.example.com",
        browser_profile_id="bp_490705123456789012",
        wait_for_completion=True,
    )

    print(f"Downloaded files: {result.downloaded_files}")

asyncio.run(main())
```

<Note>
You cannot use both `browser_profile_id` and `browser_session_id` in the same request. Choose one or the other.
</Note>

## Best practices

### Use descriptive names

```python
# Good—identifies account, site, and purpose
profile = await client.browser_profiles.create_browser_profile(
    name="prod-salesforce-admin",
    description="Admin login for daily opportunity sync",
    workflow_run_id=run_id,
)

# Bad—unclear what this is for
profile = await client.browser_profiles.create_browser_profile(
    name="profile1",
    workflow_run_id=run_id,
)
```

### Refresh profiles periodically

Session tokens and cookies expire. Re-run your login workflow and create fresh profiles before expiration:

```python
from datetime import date

# Create dated profile after each successful login
profile = await client.browser_profiles.create_browser_profile(
    name=f"crm-login-{date.today()}",
    workflow_run_id=new_login_run.run_id,
)

# Delete old profile
await client.browser_profiles.delete_browser_profile(old_profile_id)
```

### Combine with persist_browser_session

When using a profile, also set `persist_browser_session=True` to capture any state changes during the run:

```python
# Use existing profile AND save updated state
result = await client.run_workflow(
    workflow_id="wf_daily_sync",
    browser_profile_id="bp_current",
    persist_browser_session=True,  # Capture any token refreshes
    wait_for_completion=True,
)

# Create updated profile if needed
if should_refresh_profile:
    new_profile = await client.browser_profiles.create_browser_profile(
        name=f"daily-sync-{date.today()}",
        workflow_run_id=result.run_id,
    )
```
